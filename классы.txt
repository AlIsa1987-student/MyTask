' Класс Student представляет студента, сдающего нормативы
class Student {
  ' --- Поля класса (атрибуты) ---
  - FIO: string           ' Фамилия Имя Отчество студента
  - age: int              ' Возраст студента
  - group: string         ' Номер учебной группы
  
  ' --- Конструктор ---
  ' Создает объект студента с заданными параметрами
  + Student(f: string, a: int, g: string)
  
  ' --- Методы (поведение) ---
  + runs(): void          ' Метод Бег
  + jumping(): void       ' Метод Прыжки
  + pullUP(): void        ' Метод Подтягивания
  + pushUP(): void        ' Метод Отжимания
  + crouches(): void      ' Метод Приседания
  + swims(): void         ' Метод Плавание
  + dances(): void        ' Метод Танцы
  + sings(): void         ' Метод Пение
  + flexibility(): void   ' Метод Гибкость
  + sambo(): void         ' Метод Самбо
  + throwstheball(): void ' Метод Бросание мяча
  + displayInfo(): void   ' Метод для отображения информации о студенте
}

' Класс PhysEdSystem представляет систему учета нормативов
class PhysEdSystem {
  ' --- Поля класса (атрибуты) ---
  - systemName: string    ' Название системы
  
  ' --- Конструктор ---
  ' Создает объект системы с заданным именем и версией
  + PhysEdSystem(name: string, v: string)
  
  + addStudent(student: Student): void  ' Добавление студента в систему
  + simulateTesting(): void             ' Процесс сдачи нормативов
  + printResults(): void                ' Вывод результатов тестирования
}

' Ассоциация: система содержит студентов
PhysEdSystem --> Student : управляет

' Объект system представляет конкретную систему учета
object "Система Физкультура" as system {
  systemName = "Система учета нормативов"
  version = "1.0"
}
